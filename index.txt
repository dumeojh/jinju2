<script>
    /** [기능 1] 부정행위 방지: 다중 탭 접속 차단 **/
    const examChannel = new BroadcastChannel('anseong_exam_channel');
    examChannel.onmessage = function(e) {
        if (e.data === 'check_open') examChannel.postMessage('already_open');
        else if (e.data === 'already_open') {
            alert("🚨 부정행위 방지: 이미 다른 탭에서 시험이 열려 있습니다.");
            document.body.innerHTML = "<div style='text-align:center; margin-top:100px;'><h2>🚫 중복 접속이 차단되었습니다.</h2></div>";
        }
    };
    examChannel.postMessage('check_open');

    /** [기능 2] 환경 설정 및 초기화 **/
    const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwD0_5Uw88zfgJeIDblIN2vCLHinI8FFGGDP8PpxS3m5cy3ZFfRaZ5zaTIQGbbWGMkp/exec";
    
    const grid = document.getElementById('dataInputGrid');
    for(let i=1; i<=10; i++) {
        grid.innerHTML += `<div class="data-item">#${i} <input type="number" class="ix" placeholder="x"> <input type="number" class="iy" placeholder="y"></div>`;
    }

    /** [기능 3] 타이머 설정 (40분) **/
    let examTime = 45 * 60;
    let timerInterval;

    function startTimer() {
        const timerDisplay = document.getElementById('timerDisplay');
        
        timerInterval = setInterval(() => {
            let minutes = Math.floor(examTime / 60);
            let seconds = examTime % 60;
            
            minutes = minutes < 10 ? "0" + minutes : minutes;
            seconds = seconds < 10 ? "0" + seconds : seconds;
            
            timerDisplay.innerText = `⏳ 남은 시간: ${minutes}:${seconds}`;
            
            if (examTime === 5 * 60) {
                alert("⚠️ 시험 종료 5분 전입니다. 제출을 준비해 주세요.");
                timerDisplay.classList.add("blink-anim");
            }

            if (examTime <= 0) {
                clearInterval(timerInterval);
                timerDisplay.innerText = "⏳ 시험 종료";
                timerDisplay.classList.remove("blink-anim");
                alert("⏰ 시험 시간이 종료되었습니다! 현재까지 작성된 내용으로 자동 제출됩니다.");
                
                saveToSheets(true); 
            }
            
            examTime--;
        }, 1000);
    }

    /** [기능 4] 시험 시작 **/
    function startExam() {
        const cls = document.getElementById('gradeClass').value;
        const num = document.getElementById('stuNum').value;
        const name = document.getElementById('stuName').value;
        const area = document.getElementById('funcArea').value;
        
        if(!cls || !num || !name || !area) { alert("항목을 모두 입력하세요."); return; }
        
        const studentKey = `submitted_${cls.replace('2-', '')}_${num}`;
        if(localStorage.getItem(studentKey)) {
            alert("🚨 이미 제출이 완료된 학번입니다. 중복 제출은 불가능합니다.");
            return;
        }
        
        document.getElementById('authSection').style.display = 'none';
        document.getElementById('mainContent').style.display = 'block';
        document.getElementById('stickyHeader').style.display = 'block';
        document.getElementById('displayInfo').innerText = `${cls} ${num}번 ${name} (영역: ${area})`;
        
        startTimer();
    }

    /** [기능 5] 회귀 분석 및 그래프 그리기 **/
    let chartInstance = null;

    function runAnalysis() {
        const area = document.getElementById('funcArea').value;
        let pts = [];
        const xs = document.querySelectorAll('.ix'), ys = document.querySelectorAll('.iy');
        for(let i=0; i<10; i++) if(xs[i].value && ys[i].value) pts.push([parseFloat(xs[i].value), parseFloat(ys[i].value)]);
        
        if(pts.length < 3) { alert("데이터를 3개 이상 입력하세요."); return; }
        pts.sort((a,b) => a[0] - b[0]);

        let res;
        if(area === '지수함수') res = regression.exponential(pts);
        else if(area === '로그함수') res = regression.logarithmic(pts);
        else res = regression.polynomial(pts, {order:2});

        const a = res.equation[0].toFixed(4);
        const b = res.equation[1].toFixed(4);
        let latexFormula = "";

        if(area === '지수함수') latexFormula = `\\( y = ${a} \\cdot e^{${b}x} \\)`;
        else if(area === '로그함수') latexFormula = `\\( y = ${a} + ${b} \\ln(x) \\)`;
        else latexFormula = `\\( y = ${a}x^2 + ${res.equation[1].toFixed(4)}x + ${res.equation[2].toFixed(4)} \\)`;

        const formulaDiv = document.getElementById('outFormula');
        formulaDiv.innerHTML = `<b>산출된 함수식</b><br>${latexFormula}`;
        MathJax.typesetPromise([formulaDiv]);

        document.getElementById('analysisArea').style.display = 'block';
        drawChart(pts, res);
        window.scrollTo(0, document.body.scrollHeight);
    }

    function drawChart(pts, res) {
        const ctx = document.getElementById('mainChart').getContext('2d');
        if(chartInstance) chartInstance.destroy();
        const minX = Math.min(...pts.map(p=>p[0])), maxX = Math.max(...pts.map(p=>p[0]));
        let trend = [];
        for(let x=minX; x<=maxX; x+=(maxX-minX)/50) trend.push({x:x, y:res.predict(x)[1]});
        chartInstance = new Chart(ctx, {
            data: {
                datasets: [
                    { type:'scatter', label:'수집 데이터', data:pts.map(p=>({x:p[0], y:p[1]})), backgroundColor:'#3498db', pointRadius:5 },
                    { type:'line', label:'추세선', data:trend, borderColor:'#e74c3c', borderWidth:3, pointRadius:0, fill:false }
                ]
            },
            options: { scales: { x:{type:'linear', position:'bottom'}, y:{type:'linear'} } }
        });
    }

    /** [기능 6] 구글 시트에 모든 데이터 최종 제출 **/
    async function saveToSheets(isForceSubmit = false) {
        const cls = document.getElementById('gradeClass').value.replace('2-', '');
        const num = document.getElementById('stuNum').value;
        const studentKey = `submitted_${cls}_${num}`;

        if(localStorage.getItem(studentKey)) {
            alert("🚨 이미 제출이 완료된 학번입니다. 중복 제출은 불가능합니다.");
            return;
        }

        const btn = document.getElementById('saveBtn');
        
        if(!isForceSubmit) {
            if(!confirm("최종 제출하시겠습니까? 제출 후에는 수정이 불가능합니다.")) return;
        }
        
        btn.disabled = true; btn.innerText = "제출 중...";

        let ptsArr = [];
        const xs = document.querySelectorAll('.ix'), ys = document.querySelectorAll('.iy');
        for(let i=0; i<10; i++) {
            if(xs[i].value && ys[i].value) ptsArr.push(`(${xs[i].value}, ${ys[i].value})`);
        }
        
        const refs = document.querySelectorAll('.ref');

        const payload = {
            gradeClass: cls,
            stuNum: num,
            stuName: document.getElementById('stuName').value,
            funcArea: document.getElementById('funcArea').value,
            articleTitle: document.getElementById('articleTitle').value,
            ref1: refs[0].value,
            ref2: refs[1].value,
            ref3: refs[2].value,
            reason: document.getElementById('reason').value,
            dataPoints: ptsArr.join(', '),
            formula: document.getElementById('outFormula').innerText.replace("산출된 함수식", "").trim(),
            obs1: document.getElementById('obs1').value,
            obs2: document.getElementById('obs2').value,
            obs3: document.getElementById('obs3').value
        };

        try {
            await fetch(GOOGLE_SCRIPT_URL, {
                method: 'POST',
                mode: 'no-cors',
                body: JSON.stringify(payload)
            });
            
            localStorage.setItem(studentKey, 'true');
            
            alert("✅ 성공적으로 제출되었습니다!");
            btn.innerText = "제출 완료"; btn.style.background = "#2ecc71";
            btn.onclick = null;
            
            clearInterval(timerInterval);
            document.getElementById('timerDisplay').innerText = "✅ 제출 완료";
            
        } catch (e) {
            alert("❌ 오류가 발생했습니다. 화면을 캡처한 후 선생님께 문의하세요.");
            btn.disabled = false; btn.innerText = "다시 제출";
        }
    }
